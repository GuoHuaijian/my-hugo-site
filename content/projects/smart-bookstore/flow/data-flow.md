# 数据流转图

> 请求在各服务间的流转过程

## 1. 数据流转概述

### 1.1 数据流转的重要性
在分布式系统中，数据流转是系统的核心，它决定了：
- **系统性能**：数据流转效率直接影响系统吞吐量
- **数据一致性**：确保数据在各节点间正确同步
- **系统可靠性**：保证数据不丢失、不错乱
- **可观测性**：通过流转轨迹排查问题

### 1.2 数据流转原则
1. **数据完整性**：确保数据在流转过程中不丢失
2. **传输安全性**：敏感数据加密传输
3. **处理实时性**：关键数据实时流转，非关键可异步
4. **可追溯性**：每个数据包都有唯一标识

### 3.2 数据格式示例

```json
// 秒杀请求
{
  "traceId": "1234567890",
  "userId": 1001,
  "productId": 2001,
  "activityId": 3001,
  "quantity": 1,
  "clientIp": "192.168.1.100"
}

// MQ消息格式
{
  "messageId": "msg_123456",
  "traceId": "1234567890",
  "orderId": "ORDER123456789",
  "userId": 1001,
  "productId": 2001,
  "quantity": 1,
  "amount": 99.00,
  "timestamp": 1640995200000
}
```

#### 响应数据格式
```json
// 成功响应
{
  "code": "SUCCESS",
  "message": "操作成功",
  "data": {
    "orderId": "ORDER123456789",
    "status": "QUEUING",
    "queuePosition": 5,
    "estimatedTime": 30
  },
  "traceId": "1234567890"
}

// 失败响应
{
  "code": "STOCK_OUT",
  "message": "商品已售罄",
  "data": null,
  "traceId": "1234567890"
}
```

@Service
public class DataFlowTracker {

    @Autowired
    private Tracer tracer;

    /**
     * 追踪数据流转
     */
    public void trackDataFlow(DataFlow data) {
        // 生成流转ID
        String flowId = UUID.randomUUID().toString();

        // 创建追踪Span
        ScopedSpan span = tracer.nextSpan()
            .name("data-flow-" + data.getType())
            .tag("flowId", flowId)
            .tag("source", data.getSource())
            .tag("target", data.getTarget())
            .start();

        try (ActiveSpan aSpan = span.makeCurrent()) {
            // 记录流转开始
            recordFlowStart(flowId, data);

            // 处理数据流转
            processDataFlow(data);

            // 记录流转结束
            recordFlowEnd(flowId, data);

        } catch (Exception e) {
            span.error(e);
            throw e;
        } finally {
            span.finish();
        }
    }

    /**
     * 记录流转开始
     */
    private void recordFlowStart(String flowId, DataFlow data) {
        FlowRecord record = new FlowRecord();
        record.setFlowId(flowId);
        record.setType(data.getType());
        record.setSource(data.getSource());
        record.setTarget(data.getTarget());
        record.setStatus("START");
        record.setData(data.getData());
        record.setTimestamp(System.currentTimeMillis());
        flowRecordMapper.insert(record);
    }

    /**
     * 查找流转链路
     */
    public List<FlowRecord> findFlowChain(String flowId) {
        return flowRecordMapper.findByFlowId(flowId);
    }

    /**
     * 生成流转报告
     */
    public FlowReport generateFlowReport(String flowId) {
        List<FlowRecord> chain = findFlowChain(flowId);
        if (chain.isEmpty()) {
            return null;
        }

        FlowReport report = new FlowReport();
        report.setFlowId(flowId);
        report.setType(chain.get(0).getType());
        report.setSource(chain.get(0).getSource());
        report.setTarget(chain.get(chain.size() - 1).getTarget());

        // 计算总耗时
        long startTime = chain.get(0).getTimestamp();
        long endTime = chain.get(chain.size() - 1).getTimestamp();
        report.setTotalDuration(endTime - startTime);

        // 分析各环节耗时
        Map<String, Long> segmentDurations = new HashMap<>();
        for (int i = 1; i < chain.size(); i++) {
            FlowRecord prev = chain.get(i - 1);
            FlowRecord curr = chain.get(i);
            long duration = curr.getTimestamp() - prev.getTimestamp();
            segmentDurations.put(curr.getTarget(), duration);
        }
        report.setSegmentDurations(segmentDurations);

        return report;
    }
}

     */

    }

     * 批量缓存
     */
    public void batchCache(Map<String, Object> dataMap, long expireTime) {

        dataMap.forEach((key, value) -> {
            String cacheKey = "hot_data:" + key;

        });
    }
}

@Service
public class AsyncDataFlowService {

    @Async
    public CompletableFuture<Void> asyncProcessData(DataFlow data) {
        return CompletableFuture.runAsync(() -> {
            try {
                // 异步处理数据
                processData(data);
            } catch (Exception e) {
                // 记录错误
                log.error("异步处理失败", e);
            }
        });
    }

    /**
     * 批量异步处理
     */
    public CompletableFuture<Void> batchAsyncProcess(List<DataFlow> dataList) {
        List<CompletableFuture<Void>> futures = new ArrayList<>();

        for (DataFlow data : dataList) {
            futures.add(asyncProcessData(data));
        }

        // 等待所有任务完成
        return CompletableFuture.allOf(futures.toArray(new CompletableFuture[0]));
    }
}
